   104→function roundRectPath(
   105→  ctx: CanvasRenderingContext2D,
   106→  x: number,
   107→  y: number,
   108→  w: number,
   109→  h: number,
   110→  r: number
   111→) {
   112→  const rad = Math.max(0, Math.min(r, w / 2, h / 2))
   113→  ctx.beginPath()
   114→  ctx.moveTo(x + rad, y)
   115→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   116→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   117→  ctx.arcTo(x, y + h, x, y, rad)
   118→  ctx.arcTo(x, y, x + w, y, rad)
   119→  ctx.closePath()
   120→}

   122→function shade(hex: string, amt: number): string {
   123→  const n = parseInt(hex.slice(1), 16)
   124→  const r = Math.max(0, Math.min(255, ((n >> 16) & 255) + amt))
   125→  const g = Math.max(0, Math.min(255, ((n >> 8) & 255) + amt))
   126→  const b = Math.max(0, Math.min(255, (n & 255) + amt))
   127→  return `rgb(${r},${g},${b})`
   128→}

   131→function drawCar(
   132→  ctx: CanvasRenderingContext2D,
   133→  cx: number,
   134→  cy: number,
   135→  w: number,
   136→  h: number,
   137→  color: string,
   138→  isPlayer: boolean
   139→) {
   140→  const x = cx - w / 2
   141→  const y = cy - h / 2
   142→
   143→  // চাকা
   144→  ctx.fillStyle = '#0a0a0c'
   145→  const ww = w * 0.16
   146→  const wh = h * 0.17
   147→  roundRectPath(ctx, x - ww * 0.35, y + h * 0.14, ww, wh, 3)
   148→  ctx.fill()
   149→  roundRectPath(ctx, x + w - ww * 0.65, y + h * 0.14, ww, wh, 3)
   150→  ctx.fill()
   151→  roundRectPath(ctx, x - ww * 0.35, y + h * 0.69, ww, wh, 3)
   152→  ctx.fill()
   153→  roundRectPath(ctx, x + w - ww * 0.65, y + h * 0.69, ww, wh, 3)
   154→  ctx.fill()
   155→
   156→  // বডি
   157→  const grad = ctx.createLinearGradient(x, y, x, y + h)
   158→  grad.addColorStop(0, shade(color, 40))
   159→  grad.addColorStop(0.55, color)
   160→  grad.addColorStop(1, shade(color, -55))
   161→  ctx.fillStyle = grad
   162→  roundRectPath(ctx, x, y, w, h, w * 0.24)
   163→  ctx.fill()
   164→
   165→  // কেবিন
   166→  ctx.fillStyle = 'rgba(10, 14, 24, 0.9)'
   167→  roundRectPath(ctx, x + w * 0.16, y + h * 0.26, w * 0.68, h * 0.36, w * 0.16)
   168→  ctx.fill()
   169→
   170→  // সামনের কাচ
   171→  ctx.fillStyle = 'rgba(165, 180, 205, 0.55)'
   172→  roundRectPath(ctx, x + w * 0.2, y + h * 0.28, w * 0.6, h * 0.12, w * 0.1)
   173→  ctx.fill()
   174→
   175→  // হেডলাইট (সামনে)
   176→  ctx.fillStyle = isPlayer ? '#fde68a' : '#e5e7eb'
   177→  roundRectPath(ctx, x + w * 0.1, y - h * 0.008, w * 0.2, h * 0.05, 2)
   178→  ctx.fill()
   179→  roundRectPath(ctx, x + w * 0.7, y - h * 0.008, w * 0.2, h * 0.05, 2)

   181→
   182→  // টেললাইট (পেছনে)
   183→  ctx.fillStyle = isPlayer ? '#ef4444' : '#b91c1c'
   184→  roundRectPath(ctx, x + w * 0.1, y + h * 0.96, w * 0.2, h * 0.05, 2)
   185→  ctx.fill()
   186→  roundRectPath(ctx, x + w * 0.7, y + h * 0.96, w * 0.2, h * 0.05, 2)
   187→  ctx.fill()

   190→function playSound(kind: 'coin' | 'crash', muted: boolean) {
   191→  if (muted) return
   192→  try {
   193→    const Ctx =
   194→      window.AudioContext ||
   195→      (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext
   196→    if (!Ctx) return
   197→    const ctx = new Ctx()
   198→    const osc = ctx.createOscillator()
   199→    const gain = ctx.createGain()
   200→    osc.connect(gain)
   201→    gain.connect(ctx.destination)
   202→    const now = ctx.currentTime
   203→
   204→    if (kind === 'coin') {
   205→      osc.type = 'square'
   206→      osc.frequency.setValueAtTime(880, now)
   207→      osc.frequency.setValueAtTime(1320, now + 0.06)
   208→      gain.gain.setValueAtTime(0.05, now)
   209→      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12)
   210→      osc.start(now)
   211→      osc.stop(now + 0.13)
   212→    } else {
   213→      osc.type = 'sawtooth'
   214→      osc.frequency.setValueAtTime(160, now)
   215→      osc.frequency.exponentialRampToValueAtTime(40, now + 0.35)
   216→      gain.gain.setValueAtTime(0.12, now)
   217→      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4)
   218→      osc.start(now)
   219→      osc.stop(now + 0.42)
   220→    }
   221→    setTimeout(() => ctx.close().catch(() => undefined), 700)
   222→  } catch {
   223→    /* সাউন্ড সাপোর্ট না থাকলে সমস্যা নেই */
   224→  }
   225→}

   227→function createEngine(w: number, h: number): Engine {
   228→  const margin = w * ROAD_MARGIN_RATIO
   229→  const roadW = w - margin * 2
   230→  const laneW = roadW / LANES
   231→  const carW = laneW * 0.56
   232→  const carH = carW * 1.85
   233→  return {
   234→    w,
   235→    h,
   236→    playerX: margin + laneW * 1.5,
   237→    playerY: h - carH - 26,
   238→    carW,
   239→    carH,
   240→    speed: BASE_SPEED,
   241→    distance: 0,
   242→    coins: 0,
   243→    traffic: [],
   244→    coinArr: [],
   245→    particles: [],
   246→    spawnT: 0.9,
   247→    coinT: 1.4,
   248→    roadOff: 0,
   249→    shake: 0,
   250→    hudT: 0,
   251→  }
   252→}

   260→export function RacingGame({ onExit }: RacingGameProps) {
   261→  const wrapRef = useRef<HTMLDivElement>(null)
   262→  const canvasRef = useRef<HTMLCanvasElement>(null)
   263→  const engRef = useRef<Engine | null>(null)
   264→  const phaseRef = useRef<Phase>('menu')
   265→  const inputRef = useRef({
   266→    left: false,
   267→    right: false,
   268→    pointerActive: false,
   269→    pointerX: 0,
   270→  })
   271→  const mutedRef = useRef(false)
   272→  const [phase, setPhase] = useState<Phase>('menu')
   273→  const [hud, setHud] = useState<HudState>({ score: 0, coins: 0, kmh: 0 })
   274→  const [final, setFinal] = useState<FinalState | null>(null)
   275→
   276→  /** সেরা স্কোর ও সাউন্ড সেটিং — localStorage store থেকে প্রতিক্রিয়াশীলভাবে */
   277→  const best = useStoredNumber(BEST_KEY)
   278→  const muted = useStoredNumber(SOUND_KEY) === 1
   279→
   280→  /* ইঞ্জিনের ভেতরে ব্যবহারের জন্য muted রেফে মিরর করা */
   281→  useEffect(() => {
   282→    mutedRef.current = muted
   283→  }, [muted])
   284→
   285→  const setPhaseBoth = useCallback((p: Phase) => {
   286→    phaseRef.current = p
   287→    setPhase(p)
   288→  }, [])
   289→
   290→  /* গেম শুরু */
   291→  const startGame = useCallback(() => {
   292→    const canvas = canvasRef.current
   293→    if (!canvas) return
   294→    const w = canvas.clientWidth
   295→    const h = canvas.clientHeight
   296→    engRef.current = createEngine(w, h)
   297→    setFinal(null)
   298→    setHud({ score: 0, coins: 0, kmh: Math.round(BASE_SPEED / 2.6) })
   299→    setPhaseBoth('playing')
   300→  }, [setPhaseBoth])
   301→
   302→  /* বিরতি/চালু */
   303→  const togglePause = useCallback(() => {
   304→    if (phaseRef.current === 'playing') setPhaseBoth('paused')
   305→    else if (phaseRef.current === 'paused') setPhaseBoth('playing')
   306→  }, [setPhaseBoth])
   307→
   308→  const toggleMute = useCallback(() => {
   309→    writeStoredNumber(SOUND_KEY, mutedRef.current ? 0 : 1)
   310→  }, [])
   311→
   312→  /* মেইন গেম লুপ */
   313→  useEffect(() => {
   314→    const canvas = canvasRef.current
   315→    const wrap = wrapRef.current
   316→    if (!canvas || !wrap) return
   317→    const ctx = canvas.getContext('2d')
   318→    if (!ctx) return
   319→
   320→    let raf = 0
   321→    let last = performance.now()
   322→
   323→    /* সাইজিং (DPR সহ) */
   324→    const resize = () => {
   325→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   326→      const w = wrap.clientWidth
   327→      const h = wrap.clientHeight
   328→      canvas.width = Math.round(w * dpr)
   329→      canvas.height = Math.round(h * dpr)
   330→      canvas.style.width = `${w}px`
   331→      canvas.style.height = `${h}px`
   332→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   333→      if (!engRef.current) {
   334→        engRef.current = createEngine(w, h)
   335→      } else {
   336→        const eng = engRef.current
   337→        const margin = w * ROAD_MARGIN_RATIO
   338→        const roadW = w - margin * 2
   339→        const laneW = roadW / LANES
   340→        eng.w = w
   341→        eng.h = h
   342→        eng.carW = laneW * 0.56
   343→        eng.carH = eng.carW * 1.85
   344→        eng.playerY = h - eng.carH - 26
   345→        eng.playerX = Math.max(
   346→          margin + eng.carW / 2,
   347→          Math.min(margin + roadW - eng.carW / 2, eng.playerX)
   348→        )
   349→      }
   350→    }
   351→    resize()
   352→    const ro = new ResizeObserver(resize)
   353→    ro.observe(wrap)
   354→
   355→    /* স্পন + আপডেট */
   356→    const laneCenter = (eng: Engine, lane: number) => {
   357→      const margin = eng.w * ROAD_MARGIN_RATIO
   358→      const laneW = (eng.w - margin * 2) / LANES
   359→      return margin + laneW * (lane + 0.5)
   360→    }
   361→
   362→    const spawnTraffic = (eng: Engine) => {
   363→      const margin = eng.w * ROAD_MARGIN_RATIO
   364→      const laneW = (eng.w - margin * 2) / LANES
   365→      const wantTwo = eng.speed > 470 && Math.random() < 0.35
   366→      const lanes = [0, 1, 2].sort(() => Math.random() - 0.5)
   367→      const count = wantTwo ? 2 : 1
   368→      const chosen: number[] = []
   369→      for (const lane of lanes) {
   370→        if (chosen.length >= count) break
   371→        // এন্ট্রি জোনে এই লেনে কেউ আছে কি না
   372→        const occupied = eng.traffic.some(
   373→          (t) =>
   374→            Math.abs(t.x - laneCenter(eng, lane)) < laneW * 0.55 &&
   375→            t.y < eng.carH * 2.6
   376→        )
   377→        if (!occupied) chosen.push(lane)
   378→      }
   379→      for (const lane of chosen) {
   380→        eng.traffic.push({
   381→          x: laneCenter(eng, lane),
   382→          y: -eng.carH * 1.6 - Math.random() * 90,
   383→          w: eng.carW * (0.9 + Math.random() * 0.14),
   384→          h: eng.carW * 1.85 * (0.9 + Math.random() * 0.14),
   385→          rel: eng.speed * (0.5 + Math.random() * 0.3),
   386→          color: TRAFFIC_COLORS[Math.floor(Math.random() * TRAFFIC_COLORS.length)],
   387→        })
   388→      }
   389→    }
   390→
   391→    const spawnCoin = (eng: Engine) => {
   392→      const margin = eng.w * ROAD_MARGIN_RATIO
   393→      const laneW = (eng.w - margin * 2) / LANES
   394→      const lane = Math.floor(Math.random() * LANES)
   395→      const occupied = eng.traffic.some(
   396→        (t) =>
   397→          Math.abs(t.x - laneCenter(eng, lane)) < laneW * 0.6 &&
   398→          t.y > -eng.carH * 2 &&
   399→          t.y < eng.carH * 3
   400→      )
   401→      if (occupied) return
   402→      eng.coinArr.push({
   403→        x: laneCenter(eng, lane),
   404→        y: -14,
   405→        r: laneW * 0.16,
   406→        phase: Math.random() * Math.PI * 2,
   407→      })
   408→    }
   409→
   410→    const burst = (eng: Engine, x: number, y: number, colors: string[], n: number, force: number) => {
   411→      for (let i = 0; i < n; i++) {
   412→        const a = Math.random() * Math.PI * 2
   413→        const v = force * (0.3 + Math.random() * 0.9)
   414→        eng.particles.push({
   415→          x,
   416→          y,
   417→          vx: Math.cos(a) * v,
   418→          vy: Math.sin(a) * v - force * 0.3,
   419→          life: 0.5 + Math.random() * 0.5,
   420→          max: 1,
   421→          color: colors[Math.floor(Math.random() * colors.length)],
   422→          size: 2 + Math.random() * 3.5,
   423→        })
   424→      }
   425→    }
   426→
   427→    const crash = (eng: Engine) => {
   428→      burst(
   429→        eng,
   430→        eng.playerX,
   431→        eng.playerY,
   432→        ['#fbbf24', '#f97316', '#ef4444', '#94a3b8', '#52525b'],
   433→        42,
   434→        260
   435→      )
   436→      eng.shake = 0.45
   437→      setPhaseBoth('over')
   438→      playSound('crash', mutedRef.current)
   439→      if (typeof navigator !== 'undefined' && navigator.vibrate) navigator.vibrate(180)
   440→
   441→      const score = Math.floor(eng.distance / 45) + eng.coins * 10
   442→      const prevBest = readStoredNumber(BEST_KEY, 0)
   443→      const isBest = score > prevBest
   444→      if (isBest) {
   445→        writeStoredNumber(BEST_KEY, score)
   446→      }
   447→      // মোট খেলার হিসাব
   448→      writeStoredNumber('gz-plays', readStoredNumber('gz-plays', 0) + 1)
   449→      setFinal({
   450→        score,
   451→        coins: eng.coins,
   452→        kmh: Math.round(eng.speed / 2.6),
   453→        isBest,
   454→      })
   455→    }
   456→
   457→    const update = (eng: Engine, dt: number) => {
   458→      eng.speed = Math.min(MAX_SPEED, eng.speed + ACCEL * dt)
   459→      eng.distance += eng.speed * dt
   460→      eng.roadOff = (eng.roadOff + eng.speed * dt) % 44
   461→
   462→      /* স্টিয়ারিং */
   463→      const margin = eng.w * ROAD_MARGIN_RATIO
   464→      const roadW = eng.w - margin * 2
   465→      const inp = inputRef.current
   466→      if (inp.pointerActive) {
   467→        const dx = inp.pointerX - eng.playerX
   468→        const step = POINTER_STEER * dt
   469→        eng.playerX += Math.max(-step, Math.min(step, dx))
   470→      } else if (inp.left && !inp.right) {
   471→        eng.playerX -= KEY_STEER * dt
   472→      } else if (inp.right && !inp.left) {
   473→        eng.playerX += KEY_STEER * dt
   474→      }
   475→      eng.playerX = Math.max(
   476→        margin + eng.carW / 2,
   477→        Math.min(margin + roadW - eng.carW / 2, eng.playerX)
   478→      )
   479→
   480→      /* ট্রাফিক স্পন */
   481→      eng.spawnT -= dt
   482→      if (eng.spawnT <= 0) {
   483→        spawnTraffic(eng)
   484→        const interval = Math.max(0.5, 1.2 - eng.speed / 1000)
   485→        eng.spawnT = interval * (0.7 + Math.random() * 0.6)
   486→      }
   487→
   488→      /* ট্রাফিক চলাচল */
   489→      for (let i = eng.traffic.length - 1; i >= 0; i--) {
   490→        const t = eng.traffic[i]
   491→        t.y += (eng.speed - t.rel) * dt
   492→        if (t.y > eng.h + t.h * 1.6) {
   493→          eng.traffic.splice(i, 1)
   494→          continue
   495→        }
   496→        // সংঘর্ষ (একটু দয়ালু হিটবক্স)
   497→        const px = eng.playerX
   498→        const py = eng.playerY
   499→        const pw = eng.carW * 0.72
   500→        const ph = eng.carH * 0.84
   501→        const tx = t.x
   502→        const ty = t.y
   503→        const tw = t.w * 0.74
   504→        const th = t.h * 0.84
   505→        if (
   506→          Math.abs(px - tx) < (pw + tw) / 2 &&
   507→          Math.abs(py - ty) < (ph + th) / 2
   508→        ) {
   509→          crash(eng)
   510→          return
   511→        }
   512→      }
   513→
   514→      /* কয়েন */
   515→      eng.coinT -= dt
   516→      if (eng.coinT <= 0) {
   517→        spawnCoin(eng)
   518→        eng.coinT = 1.2 + Math.random() * 1.1
   519→      }
   520→      for (let i = eng.coinArr.length - 1; i >= 0; i--) {
   521→        const c = eng.coinArr[i]
   522→        c.y += eng.speed * dt
   523→        if (c.y > eng.h + 20) {
   524→          eng.coinArr.splice(i, 1)
   525→          continue
   526→        }
   527→        const nearestX = Math.max(
   528→          eng.playerX - eng.carW * 0.42,
   529→          Math.min(c.x, eng.playerX + eng.carW * 0.42)
   530→        )
   531→        const nearestY = Math.max(
   532→          eng.playerY - eng.carH * 0.46,
   533→          Math.min(c.y, eng.playerY + eng.carH * 0.46)
   534→        )
   535→        const dx = c.x - nearestX
   536→        const dy = c.y - nearestY
   537→        if (dx * dx + dy * dy < (c.r + 7) * (c.r + 7)) {
